cmake_minimum_required(VERSION 3.20)
project(oop_lab3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(lab3_core src/array.cpp src/figure.cpp src/geometry.cpp)
target_include_directories(lab3_core PUBLIC include)
add_executable(lab3 src/main.cpp)
target_link_libraries(lab3 PRIVATE lab3_core Threads::Threads)
if(MSVC)
  target_compile_options(lab3 PRIVATE /W4 /utf-8)
  target_compile_options(lab3_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(lab3 PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(lab3_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
  endif()
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/03597a01ee50ed33e9dfd640b249b4be3799d395.zip)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
  if(NOT DEFINED FETCHCONTENT_SOURCE_DIR_GOOGLETEST AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../lab1/build/_deps/googletest-src/CMakeLists.txt")
    set(FETCHCONTENT_SOURCE_DIR_GOOGLETEST "${CMAKE_CURRENT_SOURCE_DIR}/../lab1/build/_deps/googletest-src")
  endif()
  FetchContent_MakeAvailable(googletest)
  add_executable(lab3_tests tests/tests.cpp)
  target_link_libraries(lab3_tests PRIVATE lab3_core GTest::gtest_main Threads::Threads)
  include(GoogleTest)
  gtest_discover_tests(lab3_tests)
  add_test(NAME Menu.DemoAddsFiguresAndPreservesExisting
    COMMAND ${CMAKE_COMMAND}
      "-DLAB3_EXECUTABLE=$<TARGET_FILE:lab3>"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_demo.cmake")
endif()
