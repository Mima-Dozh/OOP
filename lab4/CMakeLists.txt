cmake_minimum_required(VERSION 3.20)
project(oop_lab4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(lab4_geometry src/array.cpp src/geometry.cpp src/figure.cpp)
target_include_directories(lab4_geometry PUBLIC include)
add_executable(lab4 src/main.cpp)
target_include_directories(lab4 PRIVATE include)
target_link_libraries(lab4 PRIVATE lab4_geometry Threads::Threads)
if(MSVC)
  target_compile_options(lab4 PRIVATE /W4 /utf-8)
else()
  target_compile_options(lab4 PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
  endif()
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/03597a01ee50ed33e9dfd640b249b4be3799d395.zip)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
  if(NOT DEFINED FETCHCONTENT_SOURCE_DIR_GOOGLETEST AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../lab1/build/_deps/googletest-src/CMakeLists.txt")
    set(FETCHCONTENT_SOURCE_DIR_GOOGLETEST "${CMAKE_CURRENT_SOURCE_DIR}/../lab1/build/_deps/googletest-src")
  endif()
  FetchContent_MakeAvailable(googletest)
  add_library(lab4_array_test_types OBJECT src/array.cpp)
  target_compile_definitions(lab4_array_test_types PRIVATE ARRAY_TEST_TYPES)
  target_include_directories(lab4_array_test_types PRIVATE include tests)
  add_executable(lab4_tests tests/tests.cpp)
  target_sources(lab4_tests PRIVATE $<TARGET_OBJECTS:lab4_array_test_types>)
  target_include_directories(lab4_tests PRIVATE include)
  target_link_libraries(lab4_tests PRIVATE lab4_geometry GTest::gtest_main Threads::Threads)
  include(GoogleTest)
  gtest_discover_tests(lab4_tests)
endif()
